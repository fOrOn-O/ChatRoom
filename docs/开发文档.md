## 1. 项目概述

### 1.1 项目背景

本项目旨在构建一个面向小型团队和社区的即时通讯系统，支持文字消息、文件传输等核心功能。系统采用前后端分离架构，后端使用 Go 语言开发，前端优先支持 Web 端。

### 1.2 项目目标

- 支持 **100-500 人** 同时在线
- 消息延迟 < **200ms**
- 系统可用性 > **99.5%**
- 支持单聊、群聊、文件传输等核心功能
- 提供简洁易用的 Web 界面

### 1.3 现有代码分析

当前项目已实现基础 TCP 聊天室功能：

```
现有文件结构:
├── main.go        # 程序入口
├── server.go      # TCP 服务器，广播消息
├── user.go        # 用户连接管理
└── go.mod         # Go 模块定义
```

**现有功能**:
- ✅ TCP 连接管理
- ✅ 用户上线/下线广播
- ✅ 消息广播

**待开发功能**:

### 4.2 Channel 消息路由系统

> **设计原则**: 不用 map 遍历广播，用 Channel 实现高效消息路由

#### 4.2.1 Channel 架构设计

```go
// Hub 是消息路由中心，所有消息通过 Channel 传递
type Hub struct {
    // ========== 注册/注销 Channel ==========
    register   chan *Client    // 客户端注册
    unregister chan *Client    // 客户端注销

    // ========== 消息路由 Channel ==========
    broadcast  chan *Message   // 广播消息
    private    chan *Message   // 私聊消息
    group      chan *Message   // 群聊消息

    // ========== 状态管理 ==========
    clients    map[uint]*Client           // 用户ID -> 客户端
    groups     map[uint]map[uint]*Client  // 群ID -> {用户ID -> 客户端}

    // ========== 生命周期控制 ==========
    ctx        context.Context
    cancel     context.CancelFunc
    done       chan struct{}
}
```

#### 4.2.3 核心实现代码

```go
// Hub.Run() 是消息路由的核心，使用 select 多路复用
func (h *Hub) Run() {
    for {
        select {
        // ========== 客户端注册 ==========
        case client := <-h.register:
            h.clients[client.UserID] = client
            // 加入群组映射
            for _, groupID := range client.GroupIDs {
                if h.groups[groupID] == nil {
                    h.groups[groupID] = make(map[uint]*Client)
                }
                h.groups[groupID][client.UserID] = client
            }
            log.Printf("用户 %d 上线, 当前在线: %d", client.UserID, len(h.clients))

        // ========== 客户端注销 ==========
        case client := <-h.unregister:
            if _, ok := h.clients[client.UserID]; ok {
                delete(h.clients, client.UserID)
                // 从群组移除
                for _, groupID := range client.GroupIDs {
                    delete(h.groups[groupID], client.UserID)
                }
                close(client.Send)
                log.Printf("用户 %d 下线, 当前在线: %d", client.UserID, len(h.clients))
            }

        // ========== 私聊消息 ==========
        case msg := <-h.private:
            if target, ok := h.clients[msg.ToID]; ok {
                select {
                case target.Send <- msg:
                    // 消息发送成功
                default:
                    // 发送缓冲区满，断开连接
                    close(target.Send)
                    delete(h.clients, target.UserID)
                }
            } else {
                // 用户离线，存储离线消息
                h.storeOfflineMessage(msg)
            }

        // ========== 群聊消息 ==========
        case msg := <-h.group:
            if members, ok := h.groups[msg.ToID]; ok {
                for userID, client := range members {
                    if userID == msg.FromID {
                        continue // 不发给自己
                    }
                    select {
                    case client.Send <- msg:
                    default:
                        close(client.Send)
                        delete(h.clients, userID)
                    }
                }
            }

        // ========== 广播消息 ==========
        case msg := <-h.broadcast:
            for _, client := range h.clients {
                select {
                case client.Send <- msg:
                default:
                    close(client.Send)
                    delete(h.clients, client.UserID)
                }
            }

        // ========== 优雅关闭 ==========
        case <-h.ctx.Done():
            return
        }
    }
}
```

#### 4.3.2 实现代码

```go
// Server 启动时创建根 Context
func (s *Server) Start() error {
    // 创建可取消的 Context
    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()

    // 监听系统信号
    sigCh := make(chan os.Signal, 1)
    signal.Notify(sigCh, syscall.SIGINT, syscall.SIGTERM)

    // 创建 Hub
    hub := NewHub(ctx)
    go hub.Run()

    // 启动 HTTP Server
    httpServer := &http.Server{
        Addr:    ":8080",
        Handler: s.router,
    }

    go func() {
        <-sigCh
        log.Println("收到关闭信号，正在优雅关闭...")
        cancel()                    // 取消所有子 Context
        httpServer.Shutdown(ctx)    // 关闭 HTTP Server
    }()

    return httpServer.ListenAndServe()
}

// WebSocket 连接时创建带超时的 Context
func (h *Hub) HandleWebSocket(ctx context.Context, w http.ResponseWriter, r *http.Request) {
    conn, err := upgrader.Upgrade(w, r, nil)

    }

    // 从请求 Context 派生，添加心跳超时
    clientCtx, clientCancel := context.WithTimeout(ctx, 5*time.Minute)

    client := &Client{
        Conn:   conn,
        Send:   make(chan []byte, 256),
        ctx:    clientCtx,
        cancel: clientCancel,
    }

    h.register <- client

    go client.ReadPump(h)
    go client.WritePump()
}

// Client 关闭时自动清理
func (c *Client) ReadPump(hub *Hub) {
    defer func() {
        hub.unregister <- c
        c.Conn.Close()
        c.cancel() // 取消 Context，通知 WritePump 退出
    }()

    c.Conn.SetReadLimit(maxMessageSize)
    c.Conn.SetReadDeadline(time.Now().Add(pongWait))
    c.Conn.SetPongHandler(func(string) error {
        c.Conn.SetReadDeadline(time.Now().Add(pongWait))
        return nil
    })

    for {
        select {
        case <-c.ctx.Done():
            // Context 被取消（超时或服务器关闭）
            return
        default:
            _, message, err := c.Conn.ReadMessage()
            if err != nil {
                return
            }
            hub.HandleMessage(message)
        }
    }
}
```

### 4.4 Select 多路复用模式

> **设计原则**: 用 select 同时监听多个 Channel，实现非阻塞操作

#### 4.4.1 应用场景

| 场景 | 使用方式 | 优势 |
|------|----------|------|
| 消息发送 | select + default | 非阻塞，缓冲区满时断开 |
| 心跳检测 | select + time.After | 超时自动断开 |
| 优雅关闭 | select + ctx.Done() | 响应关闭信号 |
| 批量操作 | select + done channel | 提前终止 |

#### 4.4.2 核心实现

```go
// 非阻塞消息发送
func (c *Client) SendMessage(msg *Message) bool {
    data, _ := json.Marshal(msg)
    select {
    case c.Send <- data:
        return true
    default:
        // 缓冲区满，连接可能已断开
        return false
    }
}

// 带超时的消息发送
func (c *Client) SendMessageWithTimeout(msg *Message, timeout time.Duration) bool {
    data, _ := json.Marshal(msg)
    timer := time.NewTimer(timeout)
    defer timer.Stop()

    select {
    case c.Send <- data:
        return true
    case <-timer.C:
        // 发送超时
        return false
    case <-c.ctx.Done():
        // 连接已关闭
        return false
    }
}

// WritePump 核心循环
func (c *Client) WritePump() {
    ticker := time.NewTicker(pingPeriod)
    defer func() {
        ticker.Stop()
        c.Conn.Close()
    }()

    for {
        select {
        case message, ok := <-c.Send:
            // 设置写超时
            c.Conn.SetWriteDeadline(time.Now().Add(writeWait))
            if !ok {
                // Channel 已关闭
                c.Conn.WriteMessage(websocket.CloseMessage, []byte{})
                return
            }

            w, err := c.Conn.NextWriter(websocket.TextMessage)
            if err != nil {
                return
            }
            w.Write(message)

            // 批量发送积压消息
            n := len(c.Send)
            for i := 0; i < n; i++ {
                w.Write([]byte("\n"))
                w.Write(<-c.Send)
            }

            if err := w.Close(); err != nil {
                return
            }

        case <-ticker.C:
            // 定时发送心跳
            c.Conn.SetWriteDeadline(time.Now().Add(writeWait))
            if err := c.Conn.WriteMessage(websocket.PingMessage, nil); err != nil {
                return
            }

        case <-c.ctx.Done():
            // Context 取消，退出
            return
        }
    }
}
```

### 4.5 Sync 并发安全

> **设计原则**: 读写分离，最小化锁粒度

#### 4.5.1 锁策略选择

| 数据结构 | 读写比例 | 选择的锁 | 原因 |
|----------|----------|----------|------|
| OnlineMap | 读多写少 | RWMutex | 读操作不互斥 |
| GroupMembers | 读多写少 | RWMutex | 群成员查询频繁 |
| MessageQueue | 写多读少 | Mutex | 简单互斥即可 |
| Counter | 纯写操作 | atomic | 原子操作更高效 |

#### 4.5.2 实现代码

```go
// 用户管理器 - 使用 RWMutex
type UserManager struct {
    users map[uint]*User
    mu    sync.RWMutex
}

func (m *UserManager) Get(id uint) (*User, bool) {
    m.mu.RLock()         // 读锁，允许并发读
    defer m.mu.RUnlock()
    user, ok := m.users[id]
    return user, ok
}

func (m *UserManager) Set(id uint, user *User) {
    m.mu.Lock()          // 写锁，独占访问
    defer m.mu.Unlock()
    m.users[id] = user
}

// 在线计数器 - 使用 atomic
type OnlineCounter struct {
    count int64
}

func (c *OnlineCounter) Incr() {
    atomic.AddInt64(&c.count, 1)
}

func (c *OnlineCounter) Get() int64 {
    return atomic.LoadInt64(&c.count)
}
```

### 4.6 errgroup 并发任务管理

> **设计原则**: 并发执行多个任务，任一失败则整体失败

#### 4.6.1 应用场景

```go
import "golang.org/x/sync/errgroup"

// 发送消息 - 并发执行多个操作
func (s *MessageService) SendMessage(ctx context.Context, msg *Message) error {
    var eg errgroup.Group

    // 1. 存储消息
    eg.Go(func() error {
        return s.repo.Create(ctx, msg)
    })

    // 2. 更新未读计数
    eg.Go(func() error {
        return s.cache.IncrUnread(ctx, msg.ToID, msg.FromID)
    })

    // 3. 推送通知
    eg.Go(func() error {
        return s.hub.Push(ctx, msg)
    })

    // 等待所有任务完成，任一失败则返回错误
    return eg.Wait()
}

// 用户上线 - 并发加载数据
func (s *UserService) UserOnline(ctx context.Context, userID uint) error {
    var eg errgroup.Group

    var user *User
    var friends []*User
    var groups []*Group

    // 并发加载用户信息
    eg.Go(func() error {
        var err error
        user, err = s.repo.GetByID(ctx, userID)
        return err
    })

    // 并发加载好友列表
    eg.Go(func() error {
        var err error
        friends, err = s.friendRepo.GetFriends(ctx, userID)
        return err
    })

    // 并发加载群组列表
    eg.Go(func() error {
        var err error
        groups, err = s.groupRepo.GetUserGroups(ctx, userID)
        return err
    })

    if err := eg.Wait(); err != nil {
        return err
    }

    // 缓存用户信息
    s.cache.SetUser(ctx, user)
    s.cache.SetFriends(ctx, userID, friends)
    s.cache.SetGroups(ctx, userID, groups)

    return nil
}
```

### 4.7 与传统实现的对比

#### 4.7.1 消息广播实现对比

**传统实现 (遍历 Map)**:
```go
// 问题: 遍历期间需要持锁，阻塞其他操作
func (h *Hub) Broadcast(msg []byte) {
    h.mu.RLock()
    defer h.mu.RUnlock()
    for _, client := range h.clients {
        client.Send <- msg  // 可能阻塞
```

```go
// 优势: 非阻塞，无锁，解耦
func (h *Hub) Broadcast(msg *Message) {

        // 消息进入广播 Channel
    default:
        // 广播缓冲区满，记录日志
        log.Warn("broadcast channel full")
    }
}
```

```go
    // 1. 存储消息到数据库
    s.repo.CreateMessage(msg)
    
    // 2. 更新未读消息计数
    s.incrUnreadCount(msg.ToID, msg.ToType, msg.FromUserID)
}

// 用户上线时拉取离线消息
func (s *MessageService) PullOfflineMessages(userID uint) []*Message {
    // 1. 获取未读消息
    messages := s.repo.GetUnreadMessages(userID)
    
    // 2. 标记为已读
    s.repo.MarkAsRead(userID, messages)
    
    return messages
```

```go
// api_test.go
func TestLoginAPI(t *testing.T) {
    // Setup
    router := setupRouter()
    
    // Create test user
    createUser(t, "testuser", "password123")
    
    // Test login
    body := `{"username":"testuser","password":"password123"}`
    req := httptest.NewRequest("POST", "/api/v1/auth/login", strings.NewReader(body))
    req.Header.Set("Content-Type", "application/json")
    
    w := httptest.NewRecorder()
    router.ServeHTTP(w, req)
    
    // Assert
    assert.Equal(t, 200, w.Code)
    
    var resp Response
    json.Unmarshal(w.Body.Bytes(), &resp)
    assert.Equal(t, 0, resp.Code)
    assert.NotEmpty(t, resp.Data["token"])
}
```
